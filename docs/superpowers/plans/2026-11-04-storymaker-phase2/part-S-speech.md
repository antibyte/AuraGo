```js
// Storymaker player: the cue map. One function places cues and caption lines on the audio of a block. With measured pieces
// whose chars sum to the text length, an offset inside a piece maps proportionally over that piece's used audio from its
// at_ms, so a marker at a sentence start lands on the onset of that sentence. Otherwise the whole block is proportional over
// its duration. Offsets are UTF-16 units of the stripped text. Classic script; publishes StorymakerCore.cuemap.
(function (root) {
    'use strict';
    var SM = root.StorymakerCore || (root.StorymakerCore = {});

    // prepare(rawBlock, length) = a copy of the block with length, starts[] (cumulative UTF-16 offsets) and placed.
    function prepare(rawBlock, length) {
        var out = {}, k, pieces = Array.isArray(rawBlock.pieces) ? rawBlock.pieces : [], starts = [], sum = 0;
        for (k in rawBlock) out[k] = rawBlock[k];
        for (var i = 0; i < pieces.length; i++) {
            starts.push(sum);
            sum += pieces[i].chars;
        }
        out.pieces = pieces;
        out.duration_ms = rawBlock.duration_ms > 0 ? rawBlock.duration_ms : 0;
        out.length = length;
        out.starts = starts;
        out.placed = rawBlock.speech_ok === true && pieces.length > 0 && sum === length ? 'span' : 'proportion';
        return out;
    }

    // estimated(id, seconds, length) = the block of a text without audio.
    function estimated(id, seconds, length) {
        return { id: id, placed: 'proportion', duration_ms: seconds * 1000, length: length, pieces: [], starts: [] };
    }

    // cueMap(block, utf16Offset) = ms from the block start. An offset equal to the length is the end of the last piece.
    function cueMap(block, offset) {
        if (block.placed !== 'span') return offset / Math.max(1, block.length) * block.duration_ms;
        var pieces = block.pieces, k = pieces.length - 1;
        for (var i = 0; i < pieces.length; i++) {
            if (offset < block.starts[i] + pieces[i].chars) { k = i; break; }
        }
        var p = pieces[k], share = p.chars > 0 ? (offset - block.starts[k]) / p.chars : 0;
        return p.at_ms + share * (p.out_ms - p.in_ms);
    }

    SM.cuemap = { prepare: prepare, estimated: estimated, cueMap: cueMap };
})(typeof globalThis !== 'undefined' ? globalThis : self);
```

```js
// Storymaker player: captions. A block's text is split into captions of at most 2 lines of 42 columns (East Asian wide
// characters count 2), and their times come from the same cue map as the cues. A caption lasts at least 1.0 s, except where
// the film end cuts it. Captions are always in the report; style.captions only decides drawing. Classic script; publishes
// StorymakerCore.captions.
(function (root) {
    'use strict';
    var SM = root.StorymakerCore || (root.StorymakerCore = {});

    var MAX_LINES = 2, MAX_WIDTH = 42, MIN_MS = 1000;

    // units(text) = [{a, end, b}]: the unbreakable runs in UTF-16 offsets. [a, end) is the visible run, and b also holds the
    // whitespace after it. A break lies at whitespace or where cjkBreak allows one.
    function units(text) {
        var R = SM.ranges, out = [], a = 0, i = 0, prev = -1;
        while (i < text.length) {
            var cp = text.codePointAt(i), size = cp > 0xFFFF ? 2 : 1;
            if (R.isSpace(cp)) {
                var j = i + 1;
                while (j < text.length && R.isSpace(text.codePointAt(j))) j++;
                if (i > a) out.push({ a: a, end: i, b: j });
                a = j;
                i = j;
                prev = -1;
                continue;
            }
            if (prev >= 0 && R.cjkBreak(prev, cp)) {
                out.push({ a: a, end: i, b: i });
                a = i;
            }
            prev = cp;
            i += size;
        }
        if (i > a) out.push({ a: a, end: i, b: i });
        return out;
    }

    // hardBreak(text, u) = the unit cut into runs of at most MAX_WIDTH columns, where canBreakAfter allows. The whitespace stays
    // on the last run.
    function hardBreak(text, u) {
        var R = SM.ranges, out = [], a = u.a, w = 0, i = u.a, prev = -1;
        while (i < u.end) {
            var cp = text.codePointAt(i), size = cp > 0xFFFF ? 2 : 1, cw = R.isWide(cp) ? 2 : 1;
            if (w + cw > MAX_WIDTH && prev >= 0 && R.canBreakAfter(prev, cp)) {
                out.push({ a: a, end: i, b: i });
                a = i;
                w = 0;
            }
            w += cw;
            prev = cp;
            i += size;
        }
        out.push({ a: a, end: u.end, b: u.b });
        return out;
    }

    // split(plain) = [{a, b, lines}]: offsets in UTF-16, b exclusive, with the trailing whitespace inside the range.
    function split(plain) {
        var R = SM.ranges, text = String(plain === undefined || plain === null ? '' : plain), list = [], out = [], cur = null, lineA = 0;
        units(text).forEach(function (u) {
            if (R.width(text.slice(u.a, u.end)) > MAX_WIDTH) list = list.concat(hardBreak(text, u)); else list.push(u);
        });
        list.forEach(function (u) {
            if (cur && R.width(text.slice(lineA, u.end)) <= MAX_WIDTH) {
                cur.lines[cur.lines.length - 1] = text.slice(lineA, u.end);
                cur.b = u.b;
                return;
            }
            if (cur && cur.lines.length < MAX_LINES) cur.lines.push(text.slice(u.a, u.end));
            else {
                cur = { a: u.a, b: u.b, lines: [text.slice(u.a, u.end)] };
                out.push(cur);
            }
            lineA = u.a;
            cur.b = u.b;
        });
        return out;
    }

    // forBlock(plain, block, startMs, sceneId) = the captions of one block, in film-global ms.
    function forBlock(plain, block, startMs, sceneId) {
        var parts = split(plain), map = SM.cuemap.cueMap, starts = [];
        parts.forEach(function (p, i) {
            var s = startMs + Math.round(map(block, p.a));
            if (i > 0 && s < starts[i - 1] + MIN_MS) s = starts[i - 1] + MIN_MS;
            starts.push(s);
        });
        return parts.map(function (p, i) {
            var end = Math.max(startMs + Math.round(map(block, p.b)), starts[i] + MIN_MS);
            if (i + 1 < parts.length && end > starts[i + 1]) end = starts[i + 1];
            return { start_ms: starts[i], end_ms: end, lines: p.lines, scene_id: sceneId, block_id: block.id };
        });
    }

    // settle(all, filmMs) = the captions of all blocks, sorted. Each end is lowered to the next start and to the film end, and a
    // caption left with no length is dropped.
    function settle(all, filmMs) {
        var sorted = all.slice().sort(function (x, y) { return x.start_ms - y.start_ms || x.end_ms - y.end_ms; }), out = [];
        sorted.forEach(function (c, i) {
            var end = Math.min(c.end_ms, filmMs);
            if (i + 1 < sorted.length && sorted[i + 1].start_ms < end) end = sorted[i + 1].start_ms;
            if (c.start_ms < end) out.push({ start_ms: c.start_ms, end_ms: end, lines: c.lines, scene_id: c.scene_id, block_id: c.block_id });
        });
        return out;
    }

    // at(list, ms) = the caption active at ms in a settled list, or null. Binary search.
    function at(list, ms) {
        var lo = 0, hi = list.length - 1;
        while (lo <= hi) {
            var mid = (lo + hi) >> 1, c = list[mid];
            if (ms < c.start_ms) hi = mid - 1;
            else if (ms >= c.end_ms) lo = mid + 1;
            else return c;
        }
        return null;
    }

    SM.captions = { MAX_LINES: MAX_LINES, MAX_WIDTH: MAX_WIDTH, MIN_MS: MIN_MS, split: split, forBlock: forBlock, settle: settle, at: at };
})(typeof globalThis !== 'undefined' ? globalThis : self);
```

```js
// Storymaker player: speech and thought bubbles. layout measures the lines through SM.text and fixes the box size; place puts
// the box above an anchor and inside the safe area; draw paints the box (a tail for speech, three circles for a thought) and
// the text. A bubble fades over FADE_SECONDS (scene.createBubble); the rhythm never scales that. Classic script; publishes
// StorymakerCore.bubble.
(function (root) {
    'use strict';
    var SM = root.StorymakerCore || (root.StorymakerCore = {});

    var FADE_SECONDS = 0.12;
    // design pixels from the feet to the top of the head of a standing actor at depth scale 1: the rig height, twice ACTOR_REACH in beats.js
    var HEAD_REACH = 594;
    var TAIL_SHARE = 0.024;    // tail length, as a fraction of the canvas height
    var BORDER_SHARE = 0.002;  // outline width, as a fraction of the canvas height
    var WIDTH_SHARE = 0.6;     // widest text: 60 % of the safe width
    var WIDTH_EMS = 18;        // but at least 18 em, and never more than the safe width
    var DOT_AT = [0.3, 0.62, 0.92];
    var DOT_SIZE = [0.3, 0.2, 0.12];

    function between(v, lo, hi) { return Math.max(lo, Math.min(v, hi)); }

    // layout(ctx, lines, style, opts): opts = {canvasW, canvasH, safe, family}. lines is a list of strings; each stays its own
    // paragraph. The text layout of SM.text may be a shared memo entry, so nothing here changes it.
    function layout(ctx, lines, style, opts) {
        var W = opts.canvasW, H = opts.canvasH, safe = opts.safe || SM.stage.DEFAULT_SAFE;
        var px = (style.size > 0 ? style.size : 0.05) * H;
        var padGuess = Math.round(style.box && style.box.pad > 0 ? style.box.pad * H : 0.4 * px);
        var safeW = W * safe.w / 100;
        var maxWidth = Math.min(safeW, Math.max(WIDTH_SHARE * safeW, WIDTH_EMS * px)) - 2 * padGuess;
        var text = SM.text.layout(ctx, lines.join('\n'), style, { canvasW: W, canvasH: H, short: Math.min(W, H), maxWidth: maxWidth, family: opts.family });
        var pad = text.pad > 0 ? text.pad : padGuess, tail = Math.max(6, Math.round(TAIL_SHARE * H));
        return {
            text: text, style: style, inner: { color: style.color, align: 'center' },
            w: text.width + 2 * pad, h: text.height + 2 * pad, radius: text.radius > 0 ? text.radius : Math.round(0.3 * px),
            tail: tail, half: Math.max(3, Math.round(0.4 * tail)), border: Math.max(1, Math.round(BORDER_SHARE * H)),
            safe: { left: W * safe.x / 100, top: H * safe.y / 100, right: W * (safe.x + safe.w) / 100, bottom: H * (safe.y + safe.h) / 100 }
        };
    }

    // place(lay, anchor) = {x, y, w, h, tip}: the box above the anchor, moved as little as needed to stay in the safe area.
    function place(lay, anchor) {
        var s = lay.safe;
        var x = between(anchor.x - lay.w / 2, s.left, s.right - lay.w);
        var y = between(anchor.y - lay.tail - lay.h, s.top, s.bottom - lay.tail - lay.h);
        var bottom = y + lay.h;
        var tip = { x: between(anchor.x, x, x + lay.w), y: Math.min(s.bottom, between(anchor.y, bottom + lay.tail / 2, bottom + lay.tail * 2)) };
        return { x: x, y: y, w: lay.w, h: lay.h, tip: tip };
    }

    // the rounded box and the tail as one path, so the outline has no line across the tail
    function speechPath(ctx, b, lay) {
        var rr = Math.min(lay.radius, b.w / 2, b.h / 2), bottom = b.y + b.h, half = lay.half;
        var bx = between(b.tip.x, b.x + rr + half, b.x + b.w - rr - half);
        ctx.beginPath();
        ctx.moveTo(b.x + rr, b.y);
        ctx.lineTo(b.x + b.w - rr, b.y);
        ctx.arcTo(b.x + b.w, b.y, b.x + b.w, b.y + rr, rr);
        ctx.lineTo(b.x + b.w, bottom - rr);
        ctx.arcTo(b.x + b.w, bottom, b.x + b.w - rr, bottom, rr);
        ctx.lineTo(bx + half, bottom);
        ctx.lineTo(b.tip.x, b.tip.y);
        ctx.lineTo(bx - half, bottom);
        ctx.lineTo(b.x + rr, bottom);
        ctx.arcTo(b.x, bottom, b.x, bottom - rr, rr);
        ctx.lineTo(b.x, b.y + rr);
        ctx.arcTo(b.x, b.y, b.x + rr, b.y, rr);
        ctx.closePath();
    }

    function thoughtCircles(ctx, b, lay) {
        var bottom = b.y + b.h, from = between(b.tip.x, b.x + lay.radius, b.x + b.w - lay.radius);
        for (var i = 0; i < DOT_AT.length; i++) {
            ctx.beginPath();
            ctx.arc(from + (b.tip.x - from) * DOT_AT[i], bottom + (b.tip.y - bottom) * DOT_AT[i], Math.max(1, DOT_SIZE[i] * lay.tail), 0, Math.PI * 2);
            ctx.fill();
            ctx.stroke();
        }
    }

    // draw(ctx, lay, anchor, kind, resolve, alpha): kind is speech or thought; anchor is in canvas pixels. Returns the box, or null.
    function draw(ctx, lay, anchor, kind, resolve, alpha) {
        if (!(alpha > 0)) return null;
        var box = place(lay, anchor), style = lay.style, a = alpha > 1 ? 1 : alpha;
        ctx.globalAlpha = a;
        ctx.fillStyle = resolve(style.box && style.box.fill || '#ffffff');
        ctx.strokeStyle = resolve(style.color || '$text');
        ctx.lineWidth = lay.border;
        ctx.lineJoin = 'round';
        if (kind === 'thought') {
            thoughtCircles(ctx, box, lay);
            SM.text.roundRect(ctx, box.x, box.y, box.w, box.h, lay.radius);
        } else speechPath(ctx, box, lay);
        ctx.fill();
        ctx.stroke();
        SM.text.draw(ctx, lay.text, lay.inner, box.x + box.w / 2, box.y + box.h / 2, resolve, a);
        return box;
    }

    // anchorOf(stage, st, win) = the head point of the actor state st in canvas pixels, under the camera window win.
    function anchorOf(stage, st, win) {
        var depth = stage.depthScale(st.y, st.depth);
        return stage.toCanvas(st.x, st.y - HEAD_REACH * depth * st.scale, win);
    }

    SM.bubble = { FADE_SECONDS: FADE_SECONDS, layout: layout, place: place, draw: draw, anchorOf: anchorOf };
})(typeof globalThis !== 'undefined' ? globalThis : self);
```

```js
// Storymaker player: lip-sync. track(block) turns the per-frame levels of a measured block into mouth shapes: rest below 85,
// E, A/I and O for low, mid and high levels, each with a seeded variation, and M/B/P at a short dip. The seed is the block
// id, so a block has one track and frame n stays a pure function of n. A block without levels has no track and the mouth
// rests. mouthAt reads a speech interval of an actor at a scene time. Classic script; publishes StorymakerCore.lipsync.
(function (root) {
    'use strict';
    var SM = root.StorymakerCore || (root.StorymakerCore = {});

    var LOW = 85, MID = 128, HIGH = 171;
    var VARIATION = 0.25;
    var DIP = 40;

    function shapeFor(level, r) {
        if (level < LOW) return 'rest';
        if (level < MID) return r < VARIATION ? 'fv' : 'e';
        if (level < HIGH) return r < VARIATION ? 'e' : 'ai';
        return r < VARIATION ? 'u' : 'o';
    }

    // track(block) = [mouth name per frame]. One draw per frame, so a dip never shifts the variation of later frames.
    function track(block) {
        var levels = block && block.levels, out = [];
        if (!levels || !levels.length) return out;
        var rnd = SM.seed.random(block.id);
        for (var i = 0; i < levels.length; i++) {
            var r = rnd();
            var dip = i > 0 && i < levels.length - 1 && levels[i - 1] - levels[i] >= DIP && levels[i + 1] - levels[i] >= DIP;
            out.push(dip ? 'mbp' : shapeFor(levels[i], r));
        }
        return out;
    }

    // mouthAt(interval, t): the mouth at scene time t inside a speech interval, whose data is {block}. The track is kept on the
    // block, because it never changes and a block can hold 27000 frames.
    function mouthAt(interval, t) {
        var block = interval.data && interval.data.block;
        if (!block) return 'rest';
        if (!block.mouths) block.mouths = track(block);
        var frames = block.mouths.length, span = block.duration_ms > 0 ? block.duration_ms / 1000 : interval.t1 - interval.t0;
        if (!frames || !(span > 0)) return 'rest';
        // a block holds ceil(duration * fps) frames, so this lands on frame k at t0 + k / fps with no fps at hand
        var k = Math.floor((t - interval.t0) / span * frames + 1e-9);
        return block.mouths[k < 0 ? 0 : (k >= frames ? frames - 1 : k)];
    }

    SM.lipsync = { track: track, mouthAt: mouthAt };
})(typeof globalThis !== 'undefined' ? globalThis : self);
```
