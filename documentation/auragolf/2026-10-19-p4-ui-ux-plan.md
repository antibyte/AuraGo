# AuraGolf P4 – UI/UX: Umsetzungsplan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** AuraGolf bekommt eine neue Oberfläche. Dazu gehören ein HUD mit Lochplakette, Spielerchips, einer Stärkeanzeige am Ball, Bannern, Einblendungen und einer echten Bahnkarte, außerdem ein Kartenmenü mit Welten, Bahnraster und Spieler-Setup mit Farben. Nach jeder Bahn baut sich die Scorecard animiert auf; am Ende stehen Siegertreppchen oder Ergebniskarte. Die Einführung läuft interaktiv im Spiel, und die Gestaltung folgt dem Desktop-Theme. Physik, Bahnen und Sound bleiben unverändert.

**Architecture:** Die DOM-freie Logik liegt in `minigolf-shared.js` (Palette, Namen, Farbstopps, Bodenflächen, Qualitätsregler, ±Par) und in reinen Exporten der neuen UI-Module (`formatPower`, `placeMeter`, `laneGeometry`, `obstacleOutlines`, `rankPlayers`, `relativeToPar`, `tutorialNext`, `tutorialView`). Alle diese Exporte sind in Node getestet. `minigolf-views.js` wird zur Shell mit Kopfleiste, Bühne, Fußleiste, Panels, Fokusfalle und Symbolen. HUD, Karte, Menüs, Scorecard und Coach-Marks bindet die Shell über Fabrikfunktionen ein. `tr`, `esc` und `icon` werden hineingereicht, sodass kein UI-Modul ein anderes importiert. Das Modell speichert Runden im Format v2 mit Spielerfarbe sowie Bestwerte je Runde und je Bahn. Alle Zeiten laufen über die Uhr von `frame()`; Animationen sind reines CSS.

**Tech Stack:** Vanilla-ES-Module, three.js 0.186.1, Rapier 0.21.0, esbuild-Bundle (`scripts/build-minigolf.mjs`), Node-Testskripte (`node:assert`), Go/rod-Browser-Smoke-Test, `Intl.NumberFormat`, Canvas 2D.

```js
// Shared AuraGolf constants and pure helpers. No three.js, Rapier or DOM, so the
// model, HUD, map, scene, physics and the Node tests can all import this module.
export const MAX_NAME_LENGTH = 24;
export const WORLD_ID = /^[a-z][a-z0-9-]{0,23}$/;
export const PALETTE = Object.freeze([
    { id: 'cream', hex: '#fff0c5' }, { id: 'coral', hex: '#f6a378' },
    { id: 'sky', hex: '#89c9ef' }, { id: 'lilac', hex: '#c8a6ed' },
    { id: 'lime', hex: '#a6d86f' }, { id: 'gold', hex: '#f4c542' },
    { id: 'rose', hex: '#f28cb1' }, { id: 'teal', hex: '#4cc2b4' }
].map(Object.freeze));
export const COLOR_IDS = Object.freeze(PALETTE.map(color => color.id));
export const DEFAULT_COLORS = Object.freeze(COLOR_IDS.slice(0, 4));
// Colour stops of the 3D power ring and the HUD power meter.
export const POWER_STOPS = Object.freeze([[0, '#7fe0a6'], [0.5, '#ffd789'], [1, '#ff8f77']].map(Object.freeze));
// Floors roll; physics treats every other surface as a bouncing wall and the map draws it as a rail.
export const FLOOR_SURFACES = new Set(['grass', 'sand']);
const CONTROL = /[\x00-\x1f\x7f-\x9f]/g;
const MINUS = String.fromCharCode(0x2212);

export const colorValue = id => (PALETTE.find(color => color.id === id) || PALETTE[0]).hex;

// Keeps valid, unused requested colours; other slots get their default colour
// when free, otherwise the first free palette colour.
export function assignColors(requested) {
    const taken = new Set(), result = new Array(requested.length);
    requested.forEach((id, index) => { if (COLOR_IDS.includes(id) && !taken.has(id)) { taken.add(id); result[index] = id; } });
    for (let index = 0; index < result.length; index++) {
        if (result[index]) continue;
        const preferred = DEFAULT_COLORS[index];
        result[index] = preferred && !taken.has(preferred) ? preferred : COLOR_IDS.find(id => !taken.has(id));
        taken.add(result[index]);
    }
    return result;
}

export function cleanName(value, fallback = '') {
    const cut = text => Array.from(String(text ?? '').replace(CONTROL, '').trim()).slice(0, MAX_NAME_LENGTH).join('').trim();
    return cut(value) || cut(fallback);
}

// Case-insensitive duplicates get " 2", " 3", ...; the stem shrinks to stay within 24 code points.
export function uniqueNames(names) {
    const used = new Set();
    return names.map(name => {
        let candidate = name, suffix = 1;
        while (used.has(candidate.toLocaleLowerCase())) {
            const tail = ' ' + ++suffix;
            candidate = Array.from(name).slice(0, MAX_NAME_LENGTH - tail.length).join('').trimEnd() + tail;
        }
        used.add(candidate.toLocaleLowerCase());
        return candidate;
    });
}

export const worldIds = courses => [...new Set((courses || []).map(course => course?.world).filter(id => typeof id === 'string' && WORLD_ID.test(id)))];

export const formatToPar = (diff, parLabel) => diff === 0 ? parLabel : diff < 0 ? MINUS + Math.abs(diff) : '+' + diff;

export function powerHex(value) {
    const t = Math.max(0, Math.min(1, Number(value) || 0));
    const index = t <= POWER_STOPS[1][0] ? 0 : 1;
    const [from, a] = POWER_STOPS[index], [to, b] = POWER_STOPS[index + 1];
    const k = (t - from) / (to - from), x = parseInt(a.slice(1), 16), y = parseInt(b.slice(1), 16);
    return '#' + [16, 8, 0].map(shift => Math.round(((x >> shift) & 255) * (1 - k) + ((y >> shift) & 255) * k).toString(16).padStart(2, '0')).join('');
}

// Automatic graphics tier: starts at medium, drops to low below 38 FPS (at most
// every 6 s), returns to medium after ~10 s at >= 55 FPS and 12 s after a change.
export function createQualityGovernor() {
    let tier = 'medium', changedAt = 0, goodFor = 0, sum = 0, count = 0, fps = 0;
    return {
        get tier() { return tier; },
        get fps() { return fps; },
        reset(now = 0) { tier = 'medium'; changedAt = now; goodFor = 0; },
        frame(dt, now) {
            if (!(dt > 0 && dt < 0.5)) return null;
            sum += dt; count++;
            if (sum <= 2) return null;
            const span = sum;
            fps = count / sum; sum = 0; count = 0;
            if (tier === 'medium') {
                if (fps >= 38 || now - changedAt <= 6) return null;
                tier = 'low'; changedAt = now; goodFor = 0;
                return 'down';
            }
            goodFor = fps >= 55 ? goodFor + span : 0;
            if (goodFor < 10 || now - changedAt < 12) return null;
            tier = 'medium'; changedAt = now; goodFor = 0;
            return 'up';
        }
    };
}
```

```js
const formatters = new Map();
function numberFormat(lang, options) {
    const key = (lang || '') + JSON.stringify(options);
    if (!formatters.has(key)) {
        let format;
        try { format = new Intl.NumberFormat(lang || undefined, options); } catch (_) { format = new Intl.NumberFormat('en', options); }
        formatters.set(key, format);
    }
    return formatters.get(key);
}

// "54 %" and "≈ 11 m" in the page language; distance parts joined by no-break spaces.
export function formatPower(strength, meters, lang) {
    const percent = numberFormat(lang, { style: 'percent', maximumFractionDigits: 0 }).format(Math.round(Math.max(0, Math.min(1, strength)) * 100) / 100);
    let distance = '';
    if (Number.isFinite(meters) && meters >= 0) {
        const rounded = meters < 1 ? Math.round(meters * 10) / 10 : meters < 10 ? Math.round(meters * 2) / 2 : Math.round(meters);
        distance = '≈\xa0' + numberFormat(lang, { maximumFractionDigits: 1 }).format(rounded) + '\xa0m';
    }
    return { percent, distance, text: distance ? percent + ' · ' + distance : percent };
}

// Top-left of the meter: 56 px right and 64 px above the ball, mirrored at the
// stage edges and clamped; bottom centre when the ball is not visible.
export function placeMeter(ball, width, height, size) {
    const half = size / 2;
    if (!ball?.visible) return { x: Math.round(width / 2 - half), y: Math.round(height - size - 12) };
    let cx = ball.x + 56, cy = ball.y - 64;
    if (cx + half > width) cx = ball.x - 56;
    if (cy - half < 0) cy = ball.y + 64;
    cx = Math.max(half, Math.min(width - half, cx)); cy = Math.max(half, Math.min(height - half, cy));
    return { x: Math.round(cx - half), y: Math.round(cy - half) };
}
```

```js
import { colorValue, FLOOR_SURFACES } from './minigolf-shared.js';

const FLAT = 0.55, MARGIN = 0.5;

// Top-down lane in world x/z: floor triangles per surface, rail tops, water and the view frame.
export function laneGeometry(course) {
    const floor = {}, walls = [], min = [Infinity, Infinity], max = [-Infinity, -Infinity];
    for (const shape of course?.colliders || []) {
        if (shape?.type !== 'mesh' || !Array.isArray(shape.vertices) || !Array.isArray(shape.indices)) continue;
        const v = shape.vertices, surface = shape.surface || 'grass', isFloor = FLOOR_SURFACES.has(surface);
        const target = isFloor ? (floor[surface] ||= []) : walls;
        for (let i = 0; i + 2 < shape.indices.length; i += 3) {
            const a = shape.indices[i] * 3, b = shape.indices[i + 1] * 3, c = shape.indices[i + 2] * 3;
            const ux = v[b] - v[a], uy = v[b + 1] - v[a + 1], uz = v[b + 2] - v[a + 2], wx = v[c] - v[a], wy = v[c + 1] - v[a + 1], wz = v[c + 2] - v[a + 2];
            const ny = uz * wx - ux * wz, length = Math.hypot(uy * wz - uz * wy, ny, ux * wy - uy * wx);
            if (!(length > 0) || Math.abs(ny) / length < FLAT) continue;
            target.push(v[a], v[a + 2], v[b], v[b + 2], v[c], v[c + 2]);
            if (isFloor) for (const p of [a, b, c]) { min[0] = Math.min(min[0], v[p]); min[1] = Math.min(min[1], v[p + 2]); max[0] = Math.max(max[0], v[p]); max[1] = Math.max(max[1], v[p + 2]); }
        }
    }
    const frame = Number.isFinite(min[0])
        ? { min: [min[0] - MARGIN, min[1] - MARGIN], max: [max[0] + MARGIN, max[1] + MARGIN] }
        : { min: [course?.bounds?.min?.[0] ?? -10, course?.bounds?.min?.[2] ?? -10], max: [course?.bounds?.max?.[0] ?? 10, course?.bounds?.max?.[2] ?? 10] };
    const water = (course?.water || []).map(volume => ({ min: [volume.min[0], volume.min[2]], max: [volume.max[0], volume.max[2]] }));
    return { frame, floor, walls, water };
}

function rotate(q, [x, y, z]) {
    if (!q) return [x, y, z];
    const [qx, qy, qz, qw] = q;
    const ix = qw * x + qy * z - qz * y, iy = qw * y + qz * x - qx * z, iz = qw * z + qx * y - qy * x, iw = -qx * x - qy * y - qz * z;
    return [ix * qw + iw * -qx + iy * -qz - iz * -qy, iy * qw + iw * -qy + iz * -qx - ix * -qz, iz * qw + iw * -qz + ix * -qy - iy * -qx];
}

// One rotated rectangle [x,z × 4] per obstacle part, posed like physics.getObstacles().
export function obstacleOutlines(course, poses = []) {
    const outlines = [];
    for (const obstacle of course?.obstacles || []) {
        const pose = poses.find(item => item.id === obstacle.id) || { position: obstacle.position, rotation: obstacle.rotation || [0, 0, 0, 1] };
        const parts = Array.isArray(obstacle.colliders) && obstacle.colliders.length ? obstacle.colliders : [{ position: [0, 0, 0], halfExtents: obstacle.halfExtents }];
        for (const part of parts) {
            const [hx, , hz] = part.halfExtents, center = part.position || [0, 0, 0], outline = [];
            for (const [sx, sz] of [[-1, -1], [1, -1], [1, 1], [-1, 1]]) {
                const local = rotate(part.rotation, [sx * hx, 0, sz * hz]);
                const world = rotate(pose.rotation, [local[0] + center[0], local[1] + center[1], local[2] + center[2]]);
                outline.push(world[0] + pose.position[0], world[2] + pose.position[2]);
            }
            outlines.push(outline);
        }
    }
    return outlines;
}
```

```js
import { scoreTotal } from './minigolf-model.js';
import { colorValue, formatToPar } from './minigolf-shared.js';

export function rankPlayers(players) {
    const ranked = players.map((player, index) => ({ index, player, total: scoreTotal(player) })).sort((a, b) => a.total - b.total || a.index - b.index);
    ranked.forEach((entry, position) => { entry.place = position && entry.total === ranked[position - 1].total ? ranked[position - 1].place : position + 1; });
    return ranked;
}

// Only finished holes count: completed scores plus the current hole once the player is done.
export function relativeToPar(player, round, courses) {
    const par = id => courses.find(course => course.id === id)?.par ?? 0;
    let diff = player.scores.reduce((sum, score, index) => sum + score - par(round.courseIds[index]), 0);
    if (player.done && !round.completed && round.holeIndex < round.courseIds.length) diff += player.strokes - par(round.courseIds[round.holeIndex]);
    return diff;
}
```

```js
const FINISHED = Object.freeze({ step: null, finished: true });
const AIMING = ['aim', 'power', 'release'];
const KEYS = { aim: 'tutorial_aim_keys', power: 'tutorial_power_keys', release: 'tutorial_release_keys' };
const to = step => ({ step, finished: false });
export const tutorialStart = () => to('aim');

// Events: drag {strength}, turn, strength {strength}, cancel, shot, resolved, panel, confirm, skip.
export function tutorialNext(state, event) {
    const step = state?.step;
    if (!step || !event) return state;
    switch (event.type) {
        case 'skip': return FINISHED;
        case 'drag': return step === 'aim' ? to('power') : step === 'power' && event.strength >= 0.2 ? to('release') : state;
        case 'turn': return step === 'aim' ? to('power') : state;
        case 'strength': return step === 'power' && event.strength >= 0.2 ? to('release') : state;
        case 'shot': return step === 'done' ? FINISHED : AIMING.includes(step) ? to('watch') : state;
        case 'resolved': return step === 'watch' ? to('done') : state;
        case 'panel': case 'confirm': return step === 'done' ? FINISHED : state;
        default: return state;
    }
}

export function tutorialView(state, { panel = false, camera = false, canAim = false, ballVisible = true } = {}) {
    const step = state?.step;
    if (!step || panel) return null;
    if (step === 'watch') return { step, text: 'tutorial_watch', keys: null, anchor: 'top' };
    if (step === 'done') return { step, text: 'tutorial_done', keys: null, anchor: 'help' };
    if (!canAim) return null;
    const anchor = ballVisible ? 'ball' : 'top';
    return camera ? { step, text: 'tutorial_camera', keys: null, anchor } : { step, text: 'tutorial_' + step, keys: KEYS[step], anchor };
}
```

```js
function layoutIssues() {
    const root = host.querySelector('.mg-app'), box = root.getBoundingClientRect(), issues = [];
    const shown = element => element.getClientRects().length > 0 && getComputedStyle(element).visibility !== 'hidden';
    const name = element => element.tagName.toLowerCase() + (element.classList[0] ? '.' + element.classList[0] : '') + (element.dataset.action ? '[' + element.dataset.action + ']' : '');
    for (const element of root.querySelectorAll('button,input,select,canvas,.mg-chip,.mg-plaque,.mg-banner,.mg-toast,.mg-round-card,.mg-hole-tile,.mg-player-card,.mg-step')) {
```
